Communication Pattern
// Child dispatches a bubbling CustomEvent
this.dispatchEvent(new CustomEvent('cart:add', {
  bubbles: true,
  detail: { name, price }
}));

// Parent or document listens — no tight coupling
document.addEventListener('cart:add', (e) => {
  const { name, price } = e.detail;
  // update cart state
});
